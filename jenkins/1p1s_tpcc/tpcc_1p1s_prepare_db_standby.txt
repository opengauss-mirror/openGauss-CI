#!/bin/bash

DATE=$(date "+%m%d")
DATE_YESTERDAY=$(date -d yesterday "+%m%d")
PORT=${perf_port_prefix}${DATE}

function fn_initCleanEnv() {
  echo '*************安装前清理环境**************'
  ps -ef | grep gaussdb | grep -v 'grep' | awk '{print $2}' | xargs kill -9
  ps -u $own_usrname | grep -v PID | awk '{print $1}' | xargs -I {} kill -9 {}
  userdel -r $own_usrname
  rm -rf /home/$own_usrname
  
  #rm -rf /tmp/.s.PGSQL.${perf_port_prefix}${DATE_YESTERDAY}*
  #rm -rf /tmp/.s.PGSQL.${PORT}*
  rm -rf /tmp/.s.PGSQL.*

  ipcs -m | grep ${own_usrname} | tr -s ' ' | cut -d ' ' -f 2 | xargs -I {} ipcrm -m {}
  ipcs -s | grep ${own_usrname} | tr -s ' ' | cut -d ' ' -f 2 | xargs -I {} ipcrm -m {}

  #清理缓存
  ipcs -m | awk '$2 ~/[0-9]+/ {print $2}' | while read s; do ipcrm -m $s; done
  echo 3 > /proc/sys/vm/drop_caches

  return $?
}

function fn_createUser(){
  echo '*************创建用户和组**************'
  groupadd $own_usrname
  useradd -g $own_usrname -d /home/$own_usrname -m -s /bin/bash $own_usrname
  echo $own_usrpasswd | passwd --stdin $own_usrname

  return $?
}

function fn_getInstallPkg(){
  echo '*************获取安装包并解压**************'
  pkg_path="/home/"$own_usrname"/package"
  echo 'pkg_path: $pkg_path'
  mkdir -p $pkg_path
  cd $pkg_path

  cpu=$(lscpu | grep Architecture | awk '{print $2}')
  OS=$(cat /etc/os-release | grep PRETTY_NAME | awk -F \" '{print $2}')
  os_name=""
  os_arch=""
  if [[ ${OS} =~ ^CentOS ]]
  then
    os_name="CentOS"
    os_arch="x86"
  elif [[ ${OS} =~ ^openEuler ]]
  then
    os_version=$(cat /etc/os-release | grep PRETTY_NAME | awk -F \" '{print $2}' | awk -F ' ' '{print $2}')
    os_name="openEuler"${os_version}
    if [ $os_version == 20.03 ]; then
      if [ $cpu == aarch64 ]; then
        os_arch="arm"
      elif [ $cpu == x86_64 ]; then
        os_arch="x86_openEuler"
      fi
    elif [ $os_version == 22.03 ]; then
      os_name="openEuler22.03"
      if [ $cpu == aarch64 ]; then
        os_arch="arm_2203"
      elif [ $cpu == x86_64 ]; then
        os_arch="x86_openEuler_2203"
      fi
    fi
  fi
  
  echo "-------------------------------------------"
  echo "OS:$OS"
  echo "os_arch:$os_arch"
  echo "os_name:$os_name"
  echo "os_version:$os_version"
  echo "cpu:$cpu"
  echo "-------------------------------------------"

  #echo "wget $getpkg_path/latest/$os_arch/openGauss-Server-$DB_RELEASE_VERSION-$os_name-$cpu.tar.bz2"
  #wget $getpkg_path"/latest/"$os_arch"/openGauss-Server-$DB_RELEASE_VERSION-$os_name-$cpu.tar.bz2"
  echo "scp ogperfuser@${primary_ip}:${pkg_path}/openGauss-Server-$DB_RELEASE_VERSION-$os_name-$cpu.tar.bz2 $pkg_path"
  expect <<EOF
  set timeout -1
  spawn scp ogperfuser@${primary_ip}:${pkg_path}/openGauss-Server-$DB_RELEASE_VERSION-$os_name-$cpu.tar.bz2 $pkg_path
  expect {
      "*yes/no*" { send "yes\r"; exp_continue }
      "*assword:" { send "$pkg_root_password\n" }
      "]#" { send "\n" }
  }
  expect eof
EOF

  #cp /usr1/test502/openGauss-Server-$DB_RELEASE_VERSION-$os_name-$cpu.tar.bz2 ./
  #cp /usr1/jhb/openGauss-6.0.0-openEuler-64bit.tar.bz2 $pkg_path

  if [ -f "./openGauss-Server-$DB_RELEASE_VERSION-$os_name-$cpu.tar.bz2" ]
  then
    tar -jxf openGauss-Server-$DB_RELEASE_VERSION-$os_name-$cpu.tar.bz2
  else
    echo "下载安装包失败"
    return -1
  fi

  return $?
}

function fn_tryGetInstallPkg(){
  fn_getInstallPkg
  retry=0
  while ([ $? -ne 0 ] && [ $retry -lt 5 ])
  do
    sleep 10
    echo "test1"
    fn_getInstallPkg
    echo "test2"
    retry=`expr $retry + 1`;
  done
}


function fn_setEnv(){
  echo '*************配置env文件**************'
  echo "export LD_LIBRARY_PATH=$pkg_path/lib:$LD_LIBRARY_PATH" > /home/$own_usrname/env
  echo "export GAUSSHOME=$pkg_path" >> /home/$own_usrname/env
  echo "export PGPORT=$PORT" >> /home/$own_usrname/env
  echo "export PATH=$PATH:$pkg_path/bin" >> /home/$own_usrname/env
  chown $own_usrname:$own_usrname /home/$own_usrname/env

  return $?
}

function fn_startDN(){
  echo '*************配置core路径**************'
  mkdir -p /home/corefile
  chmod 777 -R /home/corefile
  echo 'home/corefile/core-%e-%p-%t' > /proc/sys/kernel/core_pattern
  #ulimit -c unlimited
  
  echo '*************环境准备**************'
  sh $perf_tool/perf_pre.sh
  #清理缓存
  ipcs -m | awk '$2 ~/[0-9]+/ {print $2}' | while read s; do ipcrm -m $s; done
  echo 3 > /proc/sys/vm/drop_caches

  echo '*************启动数据库**************'
  su - $own_usrname <<EOF
  source /home/$own_usrname/env
  numactl -C $server_bind gs_ctl build -D $dn_path -M standby &
  sleep 30
  
  gs_ctl query -D $dn_path
EOF
  return $?
}

function main(){
  fn_initCleanEnv
  if [ $? -ne 0 ]
  then
    echo "初始化用户失败"
    return -1
  else
    echo "初始化用户成功"
  fi

  fn_createUser
  if [ $? -ne 0 ]
  then
    echo "创建用户和组失败"
    return -1
  else
    echo "创建用户和组成功"
  fi

  fn_tryGetInstallPkg
  #fn_getInstallPkg
  if [ $? -ne 0 ]
  then
    echo "获取安装包并解压失败"
    return -1
  else
    echo "获取安装包并解压成功"
  fi

  fn_setEnv
  if [ $? -ne 0 ]
  then
    echo "配置env文件失败"
    return -1
  else
    echo "配置env文件成功"
  fi

  fn_initdb
  if [ $? -ne 0 ]
  then
    echo "初始化数据库失败"
    return -1
  else
    echo "初始化数据库成功"
  fi

  fn_startDN
  if [ $? -ne 0 ]
  then
    echo "启动数据库失败"
    return -1
  else
    echo "启动数据库成功"
  fi
}

main $@
