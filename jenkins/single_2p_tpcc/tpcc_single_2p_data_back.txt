function data_backup(){
  echo "dn_path_bak is :"$dn_path_bak
  rm -rf $dn_path_bak
  echo "cp -r $dn_path $dn_path_bak"
  date
  cp -r $dn_path $dn_path_bak
  date
  #sleep 300
  chown -R $own_usrname:$own_usrname $dn_path_bak
  return $?
}

function data_dump(){
  echo "dump dn start"
  echo "*************mv $dn_path/pg_xlog $xlog_path**************"
  date
  mv $dn_path/pg_xlog $xlog_path
  date
  #sleep 300

  echo "*************mv $dn_path/pg_location/tablespace2 $ts2_path**************"
  date
  mv $dn_path/pg_location/tablespace2 $ts2_path
  date
  #sleep 100

  echo "*************mv $dn_path/pg_location/tablespace3 $ts3_path**************"
  date
  mv $dn_path/pg_location/tablespace3 $ts3_path
  date
  #sleep 100
  
  ln -svf $xlog_path/pg_xlog $dn_path
  ln -svf $ts2_path/tablespace2 $dn_path/pg_location
  ln -svf $ts3_path/tablespace3 $dn_path/pg_location

  return $?
}

function fn_startDN(){
  echo '*************环境准备**************'
  sh $perf_tool/perf_pre.sh
  #清理缓存
  ipcs -m | awk '$2 ~/[0-9]+/ {print $2}' | while read s; do ipcrm -m $s; done
  echo 3 > /proc/sys/vm/drop_caches

  echo '*************启动数据库**************'
  su - $own_usrname <<EOF
  source /home/$own_usrname/env
  echo "numactl -C $server_bind -p 0 gs_ctl start -D $dn_path &"
  numactl -C $server_bind -p 0 gs_ctl start -D $dn_path &
  sleep 60

  echo '获取版本号'
  version=`gsql -d postgres -p $PORT -c 'select version();'`
  echo "$version"
EOF
  return $?
}

function getSpace(){
  echo '*************获取各目录大小**************'
  CURRENT_GET_SPACE_TIME=$(date "+%y%m%d-%H%M%S")
  du -sh $dn_path > start_space_${CURRENT_GET_SPACE_TIME}.log
  du -sh $dn_path/* > start_directory_space_${CURRENT_GET_SPACE_TIME}.log
  du -sh $xlog_path > start_xlog_space_${CURRENT_GET_SPACE_TIME}.log
  du -sh $ts2_path/tablespace2 > start_tablespace2_space_${CURRENT_GET_SPACE_TIME}.log
  du -sh $ts3_path/tablespace3 > start_tablespace3_space_${CURRENT_GET_SPACE_TIME}.log
  return $?
}

function main(){
  stop_node
  if [ $? -ne 0 ]
  then
    echo "数据库停止失败"
    return -1
  else
    echo "数据库停止成功"
  fi

  data_backup
  if [ $? -ne 0 ]
  then
    echo "data备份失败"
    exit 1
  else
    echo "data备份成功"
  fi

  data_dump
  if [ $? -ne 0 ]
  then
    echo "data分盘失败"
    exit 1
  else
    echo "data分盘成功"
  fi
  
  getSpace
  if [ $? -ne 0 ]
  then
    echo "获取空间失败"
    exit 1
  else
    echo "获取空间成功"
  fi
  
  fn_startDN
  if [ $? -ne 0 ]
  then
    echo "numa启动数据库失败"
    exit 1
  else
    echo "numa启动数据库成功"
  fi
  
  getSpace
}

main $@
