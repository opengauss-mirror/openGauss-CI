function run_sysbench_a_nopart(){

  ./sysbench ${read_write_lua_path} --pgsql-db=${db3} ${parameters} run > ${db3}_oltp_read_write_${DATE}.log

  echo "./sysbench "${insert_only_lua_path}" --pgsql-db="${db1}" "${parameters}" run > "${db1}"_oltp_insert_"${DATE}".log"

  ./sysbench ${insert_only_lua_path} --pgsql-db=${db2} ${parameters} run > ${db2}_oltp_insert_${DATE}.log &
